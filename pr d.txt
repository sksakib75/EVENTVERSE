    static void showRegisterDialog() {

        JTextField name = new JTextField();

        JTextField email = new JTextField();

        JPasswordField password = new JPasswordField();

        Object[] fields = {"Full Name", name, "Email", email, "Password", password};

        int result = JOptionPane.showConfirmDialog(frame, fields, "Create Organizer Account", JOptionPane.OK_CANCEL_OPTION);

        if (result != JOptionPane.OK_OPTION) return;



        String n = name.getText().trim();

        String e = email.getText().trim();

        String p = new String(password.getPassword());

        if (n.isEmpty() || e.isEmpty() || p.isEmpty()) {

            message("Please fill all fields.", true);

            return;

        }

        for (User user : users) {

            if (user.email.equalsIgnoreCase(e)) {

                message("This email is already registered.", true);

                return;

            }

        }

        users.add(new Organizer(n, e, p));

        message("An account created successfully!", false);

    }

    static class SidebarIcon implements Icon {
        private final String type;

        SidebarIcon(String type) {
            this.type = type;
        }

        public int getIconWidth() { return 25; }
        public int getIconHeight() { return 25; }

        public void paintIcon(Component c, Graphics g, int x, int y) {
            Graphics2D g2 = (Graphics2D) g.create();
            g2.setRenderingHint(RenderingHints.KEY_ANTIALIASING,
                    RenderingHints.VALUE_ANTIALIAS_ON);
            g2.setColor(Color.WHITE);
            g2.setStroke(new BasicStroke(2.5f));

            int a = x + 2;
            int b = y + 2;

            switch (type) {
                case "home":
                    Polygon house = new Polygon(
                        new int[]{a+1, a+11, a+21, a+18, a+18, a+13, a+13, a+8, a+8, a+4},
                        new int[]{b+11, b+2, b+11, b+11, b+21, b+21, b+15, b+15, b+21, b+21},
                        10
                    );
                    g2.fillPolygon(house);
                    break;

                case "plus":
                    g2.drawOval(a+2, b+2, 20, 20);
                    g2.setStroke(new BasicStroke(3f));
                    g2.drawLine(a+12, b+7, a+12, b+17);
                    g2.drawLine(a+7, b+12, a+17, b+12);
                    break;

                case "calendar":
                    g2.drawRoundRect(a+2, b+4, 20, 17, 3, 3);
                    g2.drawLine(a+2, b+9, a+22, b+9);
                    g2.drawLine(a+7, b+2, a+7, b+7);
                    g2.drawLine(a+17, b+2, a+17, b+7);
                    g2.drawLine(a+7, b+14, a+9, b+14);
                    g2.drawLine(a+15, b+14, a+17, b+14);
                    break;

                case "pin":
                    g2.drawOval(a+5, b+2, 14, 14);
                    g2.drawLine(a+5, b+12, a+12, b+22);
                    g2.drawLine(a+19, b+12, a+12, b+22);
                    g2.drawOval(a+9, b+6, 6, 6);
                    break;

                case "food":
                    g2.drawLine(a+6, b+3, a+6, b+21);
                    g2.drawLine(a+3, b+3, a+3, b+8);
                    g2.drawLine(a+9, b+3, a+9, b+8);
                    g2.drawLine(a+3, b+8, a+9, b+8);
                    g2.drawLine(a+17, b+3, a+17, b+21);
                    g2.drawArc(a+14, b+3, 7, 10, 180, 180);
                    break;

                case "settings":
                    g2.drawOval(a+6, b+6, 12, 12);
                    g2.drawOval(a+10, b+10, 4, 4);
                    g2.drawLine(a+12, b+1, a+12, b+6);
                    g2.drawLine(a+12, b+18, a+12, b+23);
                    g2.drawLine(a+1, b+12, a+6, b+12);
                    g2.drawLine(a+18, b+12, a+23, b+12);
                    break;

                case "logout":
                    g2.drawRoundRect(a+2, b+3, 11, 18, 2, 2);
                    g2.drawLine(a+10, b+12, a+22, b+12);
                    g2.drawLine(a+17, b+7, a+22, b+12);
                    g2.drawLine(a+17, b+17, a+22, b+12);
                    break;
            }

            g2.dispose();
        }
    }

    static void createEvent() {

        JTextField eventNameField = new JTextField();

        JComboBox<String> eventTypeBox = new JComboBox<>(new String[]{

                "Wedding", "Holud", "Birthday", "Seminar", "Concert",

                "University Program", "Corporate Event", "Sports Event", "Cultural Program", "Other"

        });

        JTextField dateField = new JTextField();

        JTextField guestField = new JTextField();

        JTextField scheduleField = new JTextField();



        Object[] fields = {

                "Event Name:", eventNameField,

                "Event Type:", eventTypeBox,

                "Event Date:", dateField,

                "Number of Guests:", guestField,

                "Event Schedule:", scheduleField

        };



        int result = JOptionPane.showConfirmDialog(frame, fields, "Step 1: Create New Event", JOptionPane.OK_CANCEL_OPTION, JOptionPane.PLAIN_MESSAGE);

        if (result != JOptionPane.OK_OPTION) return;



        String eventName = eventNameField.getText().trim();

        String eventType = (String) eventTypeBox.getSelectedItem();

        String date = dateField.getText().trim();

        String schedule = scheduleField.getText().trim();

        int guests;

        try { guests = Integer.parseInt(guestField.getText().trim()); }

        catch (Exception e) { message("Please enter a valid guest number.", true); return; }



        if (eventName.isEmpty() || date.isEmpty() || guests <= 0) {

            message("Please enter valid event information.", true); return;

        }



        ArrayList<Venue> suitable = new ArrayList<>();

        for (Venue v : venues) if (v.capacity >= guests) suitable.add(v);

        if (suitable.isEmpty()) { message("No suitable venue found for " + guests + " guests.", true); return; }



        Venue selectedVenue = (Venue) JOptionPane.showInputDialog(frame,

                "Step 2: Select a suitable venue:", "Venue Selection",

                JOptionPane.QUESTION_MESSAGE, null, suitable.toArray(), suitable.get(0));

        if (selectedVenue == null) return;



        if (foodPackages.isEmpty()) { message("No food package available. Ask admin to add one.", true); return; }



        FoodPackage selectedFood = (FoodPackage) JOptionPane.showInputDialog(frame,

                "Step 3: Select Food Package:\nPackage price is calculated automatically from its items.",

                "Food Package", JOptionPane.QUESTION_MESSAGE, null,

                foodPackages.toArray(), foodPackages.get(0));

        if (selectedFood == null) return;



        int customize = JOptionPane.showConfirmDialog(frame,

                "Do you want to customize this package by selecting/removing individual food items?",

                "Customize Package", JOptionPane.YES_NO_OPTION);



        ArrayList<FoodItem> finalFoodItems = new ArrayList<>();

        if (customize == JOptionPane.YES_OPTION) {

            JList<FoodItem> list = new JList<>(foodItems.toArray(new FoodItem[0]));

            list.setSelectionMode(ListSelectionModel.MULTIPLE_INTERVAL_SELECTION);

            ArrayList<Integer> indexes = new ArrayList<>();

            for (FoodItem item : selectedFood.items) indexes.add(foodItems.indexOf(item));

            int[] selectedIndexes = indexes.stream().filter(i -> i >= 0).mapToInt(Integer::intValue).toArray();

            list.setSelectedIndices(selectedIndexes);

            JScrollPane sp = new JScrollPane(list);

            sp.setPreferredSize(new Dimension(430, 280));

            int r = JOptionPane.showConfirmDialog(frame, sp,

                    "Select Items - Total price will update automatically", JOptionPane.OK_CANCEL_OPTION);

            if (r != JOptionPane.OK_OPTION) return;

            finalFoodItems.addAll(list.getSelectedValuesList());

        } else {

            finalFoodItems.addAll(selectedFood.items);

        }



        if (finalFoodItems.isEmpty()) { message("Please select at least one food item.", true); return; }



        JList<ServiceItem> serviceList = new JList<>(services.toArray(new ServiceItem[0]));

        serviceList.setSelectionMode(ListSelectionModel.MULTIPLE_INTERVAL_SELECTION);

        JScrollPane serviceScroll = new JScrollPane(serviceList);

        serviceScroll.setPreferredSize(new Dimension(450, 260));

        int serviceResult = JOptionPane.showConfirmDialog(frame, serviceScroll,

                "Step 4: Select Event Services (Optional)", JOptionPane.OK_CANCEL_OPTION);

        if (serviceResult != JOptionPane.OK_OPTION) return;

        ArrayList<ServiceItem> selectedServices = new ArrayList<>(serviceList.getSelectedValuesList());



        int ticketOption = JOptionPane.showConfirmDialog(frame,

                "Is this a ticketed event?", "Ticket System", JOptionPane.YES_NO_CANCEL_OPTION);

        if (ticketOption == JOptionPane.CANCEL_OPTION || ticketOption == JOptionPane.CLOSED_OPTION) return;

        boolean ticketed = ticketOption == JOptionPane.YES_OPTION;

        double ticketPrice = 0;

        int ticketQuantity = 0;

        if (ticketed) {

            JTextField price = new JTextField();

            JTextField qty = new JTextField();

            int tr = JOptionPane.showConfirmDialog(frame,

                    new Object[]{"Ticket Price:", price, "Ticket Quantity:", qty},

                    "Ticket Details", JOptionPane.OK_CANCEL_OPTION);

            if (tr != JOptionPane.OK_OPTION) return;

            try {

                ticketPrice = Double.parseDouble(price.getText().trim());

                ticketQuantity = Integer.parseInt(qty.getText().trim());

                if (ticketPrice < 0 || ticketQuantity < 0) throw new Exception();

            } catch (Exception ex) { message("Invalid ticket information.", true); return; }

        }



        String[] paymentMethods = {"bKash", "Nagad", "Card", "Cash"};

        String paymentMethod = (String) JOptionPane.showInputDialog(frame,

                "Step 5: Select Payment Method:", "Payment",

                JOptionPane.QUESTION_MESSAGE, null, paymentMethods, paymentMethods[0]);

        if (paymentMethod == null) return;



        Event event = new Event(eventName, eventType, date, guests, schedule,

                selectedVenue, selectedFood, finalFoodItems, selectedServices,

                ticketed, ticketPrice, ticketQuantity, loggedInUser, paymentMethod);



        events.add(event);

        showEventSummary(event);

    }

        int r = JOptionPane.showConfirmDialog(frame, new Object[]{"Service Name", n, "Description", d, "Price", c}, "Add Service", JOptionPane.OK_CANCEL_OPTION);

        if (r != JOptionPane.OK_OPTION) return;

        try {

            services.add(new ServiceItem(n.getText().trim(), Double.parseDouble(c.getText().trim()), d.getText().trim()));

        ServiceItem s = (ServiceItem) JOptionPane.showInputDialog(frame, "Select service", "Remove Service", JOptionPane.WARNING_MESSAGE, null, services.toArray(), services.get(0));

        if (s != null) services.remove(s);

                "Food Packages: " + foodPackages.size() + "\n" +

                "Total Event Value: " + money(totalValue));

class EventVerseMark extends JComponent {
    private final int size;
    EventVerseMark(int size) {
        this.size = size;
        setPreferredSize(new Dimension(size, size));
        setMinimumSize(new Dimension(size, size));
        setMaximumSize(new Dimension(size, size));
        setOpaque(false);
    }
    @Override protected void paintComponent(Graphics g) {
        Graphics2D g2 = (Graphics2D) g.create();
        g2.setRenderingHint(RenderingHints.KEY_ANTIALIASING, RenderingHints.VALUE_ANTIALIAS_ON);
        int pad = Math.max(3, size / 12);
        int x = pad, y = pad + size / 8, w = size - 2 * pad, h = size - 2 * pad - size / 8;
        g2.setColor(new Color(25, 115, 245));
        g2.fillRoundRect(x, y, w, h, size / 7, size / 7);
        g2.setColor(Color.WHITE);
        g2.fillRoundRect(x + size / 12, y + size / 7, w - size / 6, h - size / 5, size / 15, size / 15);
        g2.setColor(new Color(25, 115, 245));
        g2.fillRect(x + size / 12, y + size / 7, w - size / 6, size / 7);
        g2.setStroke(new BasicStroke(Math.max(2f, size / 22f), BasicStroke.CAP_ROUND, BasicStroke.JOIN_ROUND));
        g2.drawLine(x + size / 3, y + size / 10, x + size / 3, y + size / 3);
        g2.drawLine(x + 2 * size / 3, y + size / 10, x + 2 * size / 3, y + size / 3);
        int cx = x + w / 2, cy = y + h / 2 + size / 9, r = size / 5;
        Polygon star = new Polygon();
        for (int i = 0; i < 10; i++) {
            double a = -Math.PI / 2 + i * Math.PI / 5;
            int rr = (i % 2 == 0) ? r : r / 2;
            star.addPoint(cx + (int)(Math.cos(a) * rr), cy + (int)(Math.sin(a) * rr));
        }
        g2.setColor(new Color(105, 50, 220));
        g2.fillPolygon(star);
        g2.dispose();
    }
}

class GradientPanel extends JPanel {

    private final Color color1;

    private final Color color2;

    private final int radius;



    GradientPanel(Color color1, Color color2, int radius) {

        this.color1 = color1;

        this.color2 = color2;

        this.radius = radius;

        setOpaque(false);

    }



    protected void paintComponent(Graphics g) {

        Graphics2D g2 = (Graphics2D) g.create();

        g2.setRenderingHint(RenderingHints.KEY_ANTIALIASING, RenderingHints.VALUE_ANTIALIAS_ON);

        g2.setPaint(new GradientPaint(0, 0, color1, getWidth(), 0, color2));

        g2.fillRoundRect(0, 0, getWidth(), getHeight(), radius, radius);

        g2.dispose();

        super.paintComponent(g);

    }

}

// ================= FOOD PACKAGE =================

class FoodPackage {

    String name;

    ArrayList<FoodItem> items = new ArrayList<>();



    FoodPackage(String name) {

        this.name = name;

    }



    void addItem(FoodItem item) {

        if (item != null && !items.contains(item)) items.add(item);

    }



    double getPerPersonCost() {

        double total = 0;

        for (FoodItem item : items) total += item.price;

        return total;

    }



    String itemsText() {

        if (items.isEmpty()) return "No items";

        StringBuilder sb = new StringBuilder();

        for (int i = 0; i < items.size(); i++) {

            sb.append(items.get(i).name);

            if (i < items.size() - 1) sb.append(", ");

        }

        return sb.toString();

    }



    String details() {

        return "Package: " + name +

                "\nItems: " + itemsText() +

                "\nAutomatic Price/Person: " + EventVerseApp.money(getPerPersonCost());

    }



    public String toString() {

        return name + " | " + EventVerseApp.money(getPerPersonCost()) + " / person";

    }

}

// ================= EVENT =================

class Event {

    String name;

    String type;

    String date;

    int guests;

    String schedule;

    Venue venue;

    FoodPackage foodPackage;

    ArrayList<FoodItem> selectedFoodItems;

    ArrayList<ServiceItem> selectedServices;

    boolean ticketed;

    double ticketPrice;

    int ticketQuantity;

    User organizer;

    String paymentMethod;



    Event(String name, String type, String date, int guests, String schedule,

          Venue venue, FoodPackage foodPackage, ArrayList<FoodItem> selectedFoodItems,

          ArrayList<ServiceItem> selectedServices, boolean ticketed,

          double ticketPrice, int ticketQuantity, User organizer, String paymentMethod) {

        this.name = name;

        this.type = type;

        this.date = date;

        this.guests = guests;

        this.schedule = schedule;

        this.venue = venue;

        this.foodPackage = foodPackage;

        this.selectedFoodItems = new ArrayList<>(selectedFoodItems);

        this.selectedServices = new ArrayList<>(selectedServices);

        this.ticketed = ticketed;

        this.ticketPrice = ticketPrice;

        this.ticketQuantity = ticketQuantity;

        this.organizer = organizer;

        this.paymentMethod = paymentMethod;

    }



    double getFoodPerPersonCost() {

        double total = 0;

        for (FoodItem item : selectedFoodItems) total += item.price;

        return total;

    }



    double getFoodCost() {

        return getFoodPerPersonCost() * guests;

    }



    double getServiceCost() {

        double total = 0;

        for (ServiceItem service : selectedServices) total += service.cost;

        return total;

    }



    double getTotalCost() {

        return venue.cost + getFoodCost() + getServiceCost();

    }



    String getFoodItemsText() {

        if (selectedFoodItems.isEmpty()) return "None";

        StringBuilder sb = new StringBuilder();

        for (FoodItem item : selectedFoodItems) {

            sb.append("• ").append(item.name).append(" - ")

                    .append(EventVerseApp.money(item.price)).append(" per person\n");

        }

        return sb.toString();

    }



    String getServicesText() {

        if (selectedServices.isEmpty()) return "None";

        StringBuilder sb = new StringBuilder();

        for (ServiceItem service : selectedServices) {

            sb.append("• ").append(service.name).append(" - ")

                    .append(EventVerseApp.money(service.cost)).append("\n");

        }

        return sb.toString();

    }



    String getSummary() {

        return "================ EVENTVERSE EVENT SUMMARY ================\n\n" +

                "Event Name: " + name + "\n" +

                "Event Type: " + type + "\n" +

                "Date: " + date + "\n" +

                "Schedule: " + schedule + "\n" +

                "Organizer: " + organizer.name + "\n" +

                "Guests: " + guests + "\n\n" +

                "---------------- VENUE ----------------\n" +

                venue.details() + "\n\n" +

                "---------------- FOOD ----------------\n" +

                "Selected Package: " + foodPackage.name + "\n" +

                getFoodItemsText() +

                "Food Cost/Person: " + EventVerseApp.money(getFoodPerPersonCost()) + "\n" +

                "Food Total (" + guests + " guests): " + EventVerseApp.money(getFoodCost()) + "\n\n" +

                "---------------- SERVICES ----------------\n" +

                getServicesText() +

                "Service Total: " + EventVerseApp.money(getServiceCost()) + "\n\n" +

                "---------------- PAYMENT ----------------\n" +

                "Payment Method: " + paymentMethod + "\n" +

                "Ticketed Event: " + (ticketed ? "Yes" : "No") + "\n" +

                (ticketed ? "Ticket Price: " + EventVerseApp.money(ticketPrice) + "\nTicket Quantity: " + ticketQuantity + "\n" : "") +

                "\n==========================================================\n" +

                "TOTAL EVENT COST: " + EventVerseApp.money(getTotalCost()) + "\n" +

                "==========================================================";

    }



    public String toString() {

        return name + " | " + type + " | " + date;

    }

}
